# Test-only dependencies. DELIBERATELY NOT in requirements.txt: the production
# image should not ship a test runner, and the Dockerfile does not COPY tests/
# either — so the deployed container could never run them anyway.
#
# ops/run-tests.sh installs these into a THROWAWAY container built from the same
# image, with the repo mounted. That keeps production lean while making the suite
# one command to run, which is what it was not: pytest was simply absent, so
# nothing had been run against the agent for weeks.
#
# Everything else the suite imports (fastapi, sqlalchemy, httpx for TestClient)
# is already a runtime dependency.
# This repo's CI installs ONLY this file (see .github/workflows/ci.yml), unlike
# the source repo where ops/run-tests.sh layered dev deps onto an image that
# already had the runtime ones. So pull the runtime set in explicitly.
-r requirements.txt

pytest==8.3.4

# Parallel test execution (2026-09-01). The CI backend job was ~10 minutes and
# 534 of those seconds were pytest alone — checkout and install are already
# cached and fast. The suite is parallel-safe by construction (in-memory
# sqlite per process, per-test tmp checkpoint paths); the one latent race it
# surfaced was the harness's own shared StaticPool connection, fixed in
# test_call_realtime_mode.rt_env, not papered over here.
pytest-xdist==3.8.0
